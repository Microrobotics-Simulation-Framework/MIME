"""The engine and the fit package import no `mime`, held by reading their source, and shown able to fail.

`engine/` is the part you can test against closed forms and reference numbers with MIME out of the loop (see
src/${package}/engine/__init__.py); `fit/` is the second such package, for a fitting or calibration harness or anything else
that should run without MIME (src/${package}/fit/__init__.py). That holds only while it stays true, so this reads every module
of both (with `ast`, never importing it) and fails on any `mime` name: `import` and `from ... import`, and a call of
`importlib.import_module("mime...")` or `__import__("mime...")` with a constant string. The second test is the control: each of
those, planted, must be found, every time. Add a folder to MIME_FREE for each further package that must not import `mime`;
delete one you do not want, and this test says so rather than pass on nothing.

What it cannot see: a module name that is computed (`importlib.import_module(prefix + "units")`, a name read from a file or
the environment), an `exec` or `eval` of source text, and an import hidden in a string that something else imports. A computed
import of `mime` in these packages passes this test; review is what catches it.
"""

from pathlib import Path

from import_rules import absolute_imports, top_level

PACKAGE = Path(__file__).resolve().parent.parent / "src" / "${package}"
#: The folders of the package that import no `mime`.
MIME_FREE = ("engine", "fit")


def mime_imports(source: str) -> list[str]:
    """Every module a source imports whose top-level name is `mime` (so `mime`, `mime.units`, `from mime import x`, and the
    constant string of `importlib.import_module("mime.units")` or `__import__("mime")`)."""
    return [name for name in absolute_imports(source) if top_level(name) == "mime"]


def test_the_mime_free_packages_import_no_mime():
    held: dict[str, list[str]] = {}
    for folder in MIME_FREE:
        modules = sorted((PACKAGE / folder).rglob("*.py"))
        assert modules, f"no module found under {PACKAGE / folder}: the test would pass on nothing (drop it from MIME_FREE if you removed it)"
        held.update({str(path.relative_to(PACKAGE)): mime_imports(path.read_text(encoding="utf-8")) for path in modules})
    assert {path: found for path, found in held.items() if found} == {}


def test_a_planted_import_is_found():
    """The control: each way to import `mime` that this test sees is found, so a green result above means something."""
    for planted in ("import mime", "import mime.units", "import mime.units as units", "from mime import units",
                    "from mime.units import N", "def f():\n    import mime",
                    'import importlib\nimportlib.import_module("mime.units")',
                    'import importlib\nunits = importlib.import_module("mime")',
                    'from importlib import import_module\nimport_module("mime.units")',
                    '__import__("mime")', 'def f():\n    return __import__("mime.units")'):
        assert mime_imports(planted), f"{planted!r} was not found"
    assert not mime_imports("import jax\nfrom jax import numpy\nimport mimesis")
    assert not mime_imports('import importlib\nimportlib.import_module("jax.numpy")\n__import__("mimesis")\nf("mime")')


def test_a_computed_name_is_the_limit_of_this_test():
    """What the test cannot see, stated and held: a computed module name is not found, so the docstring's limit is true."""
    assert mime_imports('import importlib\nimportlib.import_module("mi" + "me")') == []
