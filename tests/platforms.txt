# The tests the platforms workflow (.github/workflows/platforms.yml) runs on macOS and on native Windows: `path  # reason`,
# one test file a line, in the order they run. tests/test_platforms_selection.py holds that every file named exists and
# is a test file, and that the workflow runs this list and nothing else. A free-space experiment only: no file here
# needs the unit tube table (209 MB). The full lane runs on Linux alone.
tests/platforms/test_smoke.py  # the console script through pipes (`mime pending` on a folder and its document among them), a save and a run read as bytes, the env check's reasons
tests/env/test_run.py  # `mime run` and `mime verify` on a folder, and XLA's cache directory
tests/env/test_modal_script.py  # `mime modal` writes its script and files on a user's machine, read through the stub
tests/saved/test_document.py  # the document round trip
tests/saved/test_load.py  # load, verify and `mime verify`, and the edits a load refuses
tests/saved/test_schema.py  # the document's schema
tests/saved/test_array_guards.py  # the arrays a save refuses to hold
tests/saved/test_chain_saved.py  # description files of one name in different folders: paths joined and copied
tests/sharing/test_referenced.py  # a sharing round trip (save, pack, unpack, a referenced file placed from the cache), into a folder reached through a link and into the platform's temporary folder as it names it
tests/sharing/test_push_cli.py  # a push's round trip to the offline fake behind a local server, through the real HTTP transport: --init, the draft, the push again, --status, --discard, an upload cut and sent again, a refusal's shape
tests/test_gitattributes.py  # the line-end rules cover every tracked file, so a checkout on Windows keeps the bytes a pinned hash was taken of
tests/errors/test_corpus.py  # the error corpus: each planted mistake, short and in full, in a fresh process
tests/platforms/test_interrupt.py  # Ctrl+C on `mime run` in process: the first press acknowledged and a stop asked, the second leaves, a press before the run raises, a run stopped by a real signal is written as stopped (the same behaviour in a child process is POSIX only, tests/env/test_run_interrupt.py)
tests/cli/test_durations.py  # a duration option's forms (10s, 500ms, 2min) are read through mime.units, a bare number stays seconds
tests/cli/test_progress.py  # the arithmetic and the words of a run's progress line: the estimate leaves the compile out, a flag is "by" a step
tests/cli/test_guide.py  # grouped --help, an example under every command, a bare `mime`, "did you mean", on this platform's argparse
tests/cli/test_pipes.py  # output through a pipe is byte for byte what it always was, with a planted colour code found (every platform; a pipe is a pipe on Windows)
tests/cli/test_terminal.py  # what mime draws on a terminal, through a pseudo-terminal: macOS runs it, Windows skips it (no pty module; stated in docs/platforms.md)
tests/cli/test_push_display.py  # what mime push and mime pull draw on a terminal, through a pseudo-terminal: macOS runs it, Windows skips it (no pty module; stated in docs/platforms.md)
tests/cli/test_command_tables.py  # the tables `mime verify`, `mime env check` and `mime registry` draw on a terminal, through a pseudo-terminal (macOS runs them, Windows skips them), cell by cell; and the byte-for-byte test that their piped output is what it always was (every platform). Two verify tests need a readable CPU flags hash and skip where there is none (`needs_readable_cpu_flags`: they assert `bit for bit`, which an unreadable CPU cannot claim); the words for another environment are tested on macOS too, with the environment's difference planted
tests/assets/test_download_gates.py  # the licence gate and the disk gate in the library, on every platform: the free space patched (shutil.disk_usage is each platform's own), a download that fails part-way, the mutants caught on every run
tests/cli/test_download_gates_cli.py  # the licence panel and its question through a pseudo-terminal (macOS runs them, Windows skips them: no pty module), and the refusals through pipes with no terminal to ask at (every platform): --accept-licence names the licence, --yes and --force accept none
tests/cli/test_asset_tables.py  # the asset commands' tables and the unwrapped paste blocks on a terminal (macOS runs them, Windows skips them), and the byte-for-byte test that their piped output is what it always was (every platform)
tests/cli/test_import_interrupt.py  # Ctrl+C while the command imports exits quietly, 130 (POSIX only: SIGINT to a child)
